  function loadData(dataObj) {
    currentData = dataObj;

    for (const playerName in currentData.players) {
      const player = currentData.players[playerName];
      const earned = player.earnedCards || [];
      player.games.forEach(game => {
        game.cards = [];
        earned.forEach(ec => {
          if (ec.appliedToGame && ec.appliedToGame.toLowerCase() === game.name.toLowerCase()) {
            game.cards.push(ec.cardName);
          }
        });
      });
    }

    renderPlayerTab('Никита');
    renderPlayerTab('Вадим');
    renderPlayerTab('Катя');
    renderGeneralStats();
    renderCardsTab(currentCardPlayer);
    renderWheelGames();
    const firstTab = document.querySelector('#mainTabs .nav-link');
    if (firstTab) {
      const tab = new bootstrap.Tab(firstTab);
      tab.show();
    }
  }

  function getGames(playerName) {
    if (!currentData) return [];
    const p = currentData.players[playerName];
    return p ? p.games : [];
  }

    if (!currentData) return [];

    return p ? p.earnedCards || [] : [];
  }

  function getAllCardsDescriptions() {
    return currentData ? currentData.cards || [] : [];
  }

  function calcStats(playerName) {
    const games = getGames(playerName);
    let totalTime = 0, totalPoints = 0, totalRating = 0, ratingCount = 0;
    games.forEach(g => {
      if (g.time !== null && g.time !== undefined && !isNaN(parseFloat(g.time))) totalTime += parseFloat(g.time);
      if (g.points !== null && g.points !== undefined && !isNaN(parseFloat(g.points))) totalPoints += parseFloat(g.points);
      let numRating = parseFloat(g.rating);
      if (!isNaN(numRating)) {
        totalRating += numRating;
        ratingCount++;
      }
    });
    return {
      count: games.length,
      totalTime: totalTime,
      totalPoints: totalPoints,
      avgRating: ratingCount > 0 ? (totalRating / ratingCount) : 0
    };
  }

  function renderTable(playerName, gamesArray) {

  function filterTable(playerName) {
    const games = getGames(playerName);
    const searchVal = document.getElementById(`searchInput-${playerName}`).value.toLowerCase();
    const diffVal = document.getElementById(`filterDifficulty-${playerName}`).value;
    const filtered = games.filter(g => {
      const matchName = g.name.toLowerCase().includes(searchVal);
      const matchDiff = diffVal === '' || g.difficulty === diffVal;
      return matchName && matchDiff;
    });
    if (currentSort.player === playerName && currentSort.key) {
      sortGames(filtered, currentSort.key, currentSort.asc);
    }
    renderTable(playerName, filtered);
  }

  function sortTable(playerName, key) {
    const games = getGames(playerName);
    if (currentSort.player === playerName && currentSort.key === key) {
      currentSort.asc = !currentSort.asc;
    } else {
      currentSort.player = playerName;
      currentSort.key = key;
      currentSort.asc = true;
    }
    const searchVal = document.getElementById(`searchInput-${playerName}`).value.toLowerCase();
    const diffVal = document.getElementById(`filterDifficulty-${playerName}`).value;
    const filtered = games.filter(g => {
      const matchName = g.name.toLowerCase().includes(searchVal);
      const matchDiff = diffVal === '' || g.difficulty === diffVal;
      return matchName && matchDiff;
    });
    sortGames(filtered, key, currentSort.asc);
    renderTable(playerName, filtered);
  }

  function sortGames(arr, key, asc) {
    arr.sort((a, b) => {
      let valA = a[key];
      let valB = b[key];
      if (['time', 'points'].includes(key)) {
        valA = parseFloat(valA) || 0;
        valB = parseFloat(valB) || 0;
        return asc ? valA - valB : valB - valA;
      }
      valA = (valA || '').toString().toLowerCase();
      valB = (valB || '').toString().toLowerCase();
      if (valA < valB) return asc ? -1 : 1;
      if (valA > valB) return asc ? 1 : -1;
      return 0;
    });
  }

  function drawChart(playerName) {
    const canvas = document.getElementById(`chart-${playerName}`);
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const games = getGames(playerName);
    const labels = games.slice(0, 15).map(g => g.name.length > 15 ? g.name.slice(0, 12)+'…' : g.name);
    const dataPoints = games.slice(0, 15).map(g => {
      let val = parseFloat(g.rating);
      return isNaN(val) ? 0 : val;
    });

    if (chartInstances[playerName]) {
      chartInstances[playerName].destroy();
    }

    chartInstances[playerName] = new Chart(ctx, {
      type: 'bar',
      data: {
        labels: labels,
        datasets: [{
          label: 'Оценка',
          data: dataPoints,
          backgroundColor: '#ffb347',
          borderRadius: 4
        }]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        plugins: { legend: { display: false } },
        scales: {
          y: { beginAtZero: true, ticks: { color: '#aaa' }, grid: { color: '#555' } },
          x: { ticks: { color: '#aaa', maxRotation: 45 } }
        }
      }
    });
  }

  function formatDateOrNumber(val) {
    if (!val) return '—';
    if (typeof val === 'string' && /^\d{4}-\d{2}-\d{2}/.test(val)) {
      const d = new Date(val);
      if (!isNaN(d)) return d.toLocaleDateString('ru-RU');
    }
    return val;
  }

  function exportCSV(playerName) {
    const games = getGames(playerName);
    if (!games.length) return;
    const headers = ['Название','Сложность','Время','Оценка','HLTB','Metacritic','Баллы'];
    const rows = games.map(g => [
      g.name,
      g.difficulty || '',
      g.time !== null ? g.time : '',
      g.rating || '',
      g.hltb || '',
      formatDateOrNumber(g.metacritic),
      g.points !== null ? g.points : ''
    ]);
    let csv = headers.join(',') + '\n';
    rows.forEach(row => { csv += row.join(',') + '\n'; });
    const blob = new Blob([csv], {type: 'text/csv'});
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${playerName}_games.csv`;
    a.click();
    URL.revokeObjectURL(url);
  }

      } catch (err) {
        alert('Ошибка парсинга JSON: ' + err.message);
      }
    };
    reader.readAsText(file);
  });
